38-Like

-- Retrieves employees whose first name starts with the letter 'a'
Select ID,FirstName,LastName From Employees
where FirstName Like 'a%';

-- Retrieves employees whose first name ends with the letter 'a'
Select ID,FirstName,LastName From Employees
where FirstName Like '%a';

-- Retrieves employees whose first name contains the substring 'tell' anywhere inside it
Select ID,FirstName,LastName From Employees
where FirstName Like '%tell%';

-- Retrieves employees whose first name starts and ends with the letter 'a'
Select ID,FirstName,LastName From Employees
where FirstName Like 'a%a';

-- Retrieves employees whose first name has 'a' as the second character
Select ID,FirstName,LastName From Employees
where FirstName Like '_a%';

-- Retrieves employees whose first name has 'a' as the third character
Select ID,FirstName,LastName From Employees
where FirstName Like '__a%';

-- Retrieves employees whose first name starts with 'a' and has at least 3 characters in total
Select ID,FirstName,LastName From Employees
where FirstName Like 'a__%'

-- Retrieves employees whose first name either starts with 'a' OR has 'b' as the third character
Select ID,FirstName,LastName From Employees
where FirstName Like 'a%' or FirstName Like '__b%';

39-WildCards

-- Updates the first name to 'Mohammed' and last name to 'Ali' for the employee with ID 285
Update Employees
Set FirstName ='Mohammed' , LastName='Ali'
where ID =285;

-- Updates the first name to 'Mohammad' and last name to 'Maher' for the employee with ID 286
Update Employees
Set FirstName ='Mohammad' , LastName='Maher'
where ID =286;

-- Retrieves employees whose first name matches either 'Mohammed' or 'Mohammad' using a character set wildcard
Select Id,FirstName,LastName From Employees
where FirstName like 'Mohamm[ae]d';

-- Retrieves employees whose first name does NOT match 'Mohammed' or 'Mohammad' using a negated character set wildcard
Select Id,FirstName,LastName From Employees
where FirstName Not like 'Mohamm[ae]d';

-- Retrieves employees whose first name starts with any of the characters 'a', 'b', or 'c'
Select Id,FirstName,LastName From Employees
where FirstName  like '[abc]%';

-- Retrieves employees whose first name starts with any letter from 'A' to 'L', ordered alphabetically by first name
Select Id,FirstName,LastName From Employees
where FirstName  like '[A-L]%'
Order By FirstName;
