	const float PI = 3.14;
    const float P = (a + c + b) / 2;;
	float Result = 0;

 int  SumOddNumByForLoop(int N) {
	int Sum = 0;

	for (int Counter  = 0;Counter <=N; Counter++) {
		if (CheckEvenOrOdd (Counter)==en_EvenOrOdd::Odd)
			Sum += Counter;
	}
	return Sum;
}
int SumOddNumByWhileLoop(int N) {
	int Sum = 0;
	int Counter = 0;
	cout << "Sum Odd Num With While Loop:";
	while (Counter <= N) {
		if (CheckEvenOrOdd(Counter)==en_EvenOrOdd::Odd)
	    Sum += Counter;
		Counter++;
	}
	return Sum;
}

		if (CheckEvenOrOdd (Counter)==en_EvenOrOdd::Odd)

		Counter++;
	} while (Counter <= Num);
