	int Integer_Number = (int)Number;
	float Fraction = FractionPart(Number);
		if (abs(Fraction)>= 0.5) {
			if (Number > 0)
				return ++Integer_Number;
			else
				return --Integer_Number;
		}
		else {
			return Integer_Number;
		}
