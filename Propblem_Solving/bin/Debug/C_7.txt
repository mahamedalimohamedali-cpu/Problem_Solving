bool IsIdentity(int Matrix[3][3],short Row ,short Col){
	int Counter = 0;
	for (int i = 0; i < Row; i++) {
		for (int j = 0; j < Col; j++) {
			if (i == j && Matrix[i][j] != 1)
				return false;
			else if (i != j && Matrix[i][j] != 0)
				return false;
		}
	}
	return true;
}

int CountNumberinMatrix(int Matrix[3][3], int Row, int Col,int Number) {
	short Counter = 0;
	for (int i = 0; i < Row; i++) {
		for (int j = 0; j < Col; j++) {
			if (Matrix[i][j] == Number)
				Counter++;
		}
	}

	return Counter;
}

bool IsSparse(int Matrix[3][3],short Row ,short Col){
	int MatrixSize = Row * Col;
	return (CountNumberinMatrix(Matrix, Row, Col, 0) > (MatrixSize/2));
}

	return "";
}

	return TrimeLeft(TrimeRight(Name));
