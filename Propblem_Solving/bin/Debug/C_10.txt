15- Virtual Function  (ِشرح)

تمام يا محمد، خليني أشرحلك الموضوع بالتفصيل علشان يبقى عندك صورة كاملة وتحتفظ بيها 👇  

---

## 🧩 الفكرة الأساسية
- عندك **Base Class** فيه دوال وخصائص مشتركة بين كل الكائنات (زي الصفحات في برنامج).  
- لو فيه دالة معينة عايزها تختلف من كلاس للتاني → تعملها **virtual** في الـ Base Class.  
- كل كلاس مشتق يعمل **Override** للدالة دي ويكتب التنفيذ الخاص بيه.  
- لما تستخدم **Pointer من نوع Base Class** (`Base*`) وتشاور على أي كائن مشتق، هتقدر تنادي نفس الدالة، لكن التنفيذ هيختلف حسب النوع الحقيقي للكائن.  

---

## ⚙️ إزاي بيشتغل؟
1. **الدوال العادية (مش virtual)**:  
   - بتنادي نسخة الـ Base Class دايمًا، حتى لو الكائن مشتق.  
2. **الدوال virtual**:  
   - وقت التنفيذ (runtime)، الـ system بيشوف نوع الكائن الحقيقي اللي الـ pointer بيشاور عليه.  
   - لو فيه Override في الكلاس المشتق → ينفذ النسخة الجديدة.  
   - لو مفيش Override → يرجع ينفذ نسخة الـ Base Class.  

---

## 📖 مثال عملي (برنامج صفحات)
```cpp
class Page {
public:
    void Header() { cout << "Common Header\n"; }
    void Footer() { cout << "Common Footer\n"; }
    virtual void Render() { cout << "Base Render\n"; }
};

class HomePage : public Page {
public:
    void Render() override { cout << "Render Home Page\n"; }
};

class ContactPage : public Page {
public:
    void Render() override { cout << "Render Contact Page\n"; }
};

class AboutPage : public Page {
public:
    void Render() override { cout << "Render About Page\n"; }
};
```

```cpp
int main() {
    HomePage home;
    ContactPage contact;
    AboutPage about;

    Page* pages[] = { &home, &contact, &about };

    for (Page* p : pages) {
        p->Header();   // ثابت من الـ Base Class
        p->Render();   // مختلف حسب نوع الصفحة
        p->Footer();   // ثابت من الـ Base Class
        cout << "-----------------\n";
    }
}
```

---

## 🖼️ النتيجة
- كل الصفحات عندها **Header** و **Footer** ثابتين (من الـ Base Class).  
- لكن دالة **Render** بتختلف حسب الصفحة لأننا عملناها `virtual` وكل كلاس مشتق غيّرها بطريقته.

- الـ **virtual functions** بتخليك تكتب كود عام موحّد، لكن التنفيذ يختلف حسب الكلاس المشتق.  
- ده بيدي مرونة رهيبة في البرامج الكبيرة: نفس الواجهة، لكن التفاصيل تتغير حسب الحاجة.  
- الفكرة دي هي أساس الـ Polymorphism في الـ OOP.  

---

تحب أديك مثال تاني لكن ببرنامج **Menu System** (Home, Settings, Profile) علشان تشوف إزاي كل صفحة بتنفذ نفس الدالة بشكل مختلف؟

15- Virtaul Function

#include <iostream>
using namespace std;

class clsPerson 
{
public:
	/*virtual*/ void Print() { cout << "Hi, Iam Person!\n"; } //Without Virtual 
	virtual void Print1() { cout << "Hi, Iam Person!\n"; }
};

class clsEmp :public clsPerson 
{
public:
	void Print1() { cout << "Hi, Iam Employee!\n";}
};

class clsStudent :public clsPerson
{
public:
	void Print1() { cout << "Hi, Iam Student!\n"; }
};

int main()
{
	//Way1
	/*
	clsPerson* Person1 = new clsEmp;
	clsPerson* Person2 = new clsStudent;
	Person1->Print1();
	Person2->Print1();
	delete Person1, Person2;*/
	//Another Way
	clsEmp Emp;
	clsStudent S;
	Emp.Print1();
	S.Print1();

	clsPerson* Person1 = &Emp;
	clsPerson* Person2 = &S;
	Person1->Print1();
	Person2->Print1();

	Person1->Print();
	Person2->Print();

	system("pause>0");
}

#include <iostream>

class clsPerson

	virtual void Print1() { cout << "Hi, Iam Person!\n"; }
};

class clsEmp :public clsPerson 
{
public:
	void Print1() { cout << "Hi, Iam Employee!\n";}
};

class clsStudent :public clsPerson
{
public:
	void Print1() { cout << "Hi, Iam Student!\n"; }
};

int main()
{
	//Way1
	/*
	clsPerson* Person1 = new clsEmp;
	clsPerson* Person2 = new clsStudent;
	Person1->Print1();
	Person2->Print1();
	delete Person1, Person2;*/
	//Another Way
        //Eraly/Static Binding
	clsEmp Emp;

	S.Print1();

	Person2->Print1();

	Person1->Print();
	Person2->Print();

	system("pause>0");

	int _Var3;

	int Var2;

	}

19-Friend Function 


#include <iostream>
using namespace std;

class clsA 
{
private :
	int _Var1;
protected:
	int _Var2;
public:
	int Var3;
	clsA() 
	{
		_Var1 = 10;
		_Var2 = 20;
		 Var3 = 30;
	}

	friend int MySum(clsA A1);//Friend Function

};

int MySum(clsA A1) 
{
	return A1._Var1 + A1._Var2 + A1.Var3;
}
int main()
{
	clsA A1;

	cout << "Sum Of Members :" << MySum(A1);

	system("pause>0");
}

23-Passing Objects to Functions (ByRef/ByVal)

#include <iostream>
using namespace std;
class clsA 
{
public:
	int _X;
	clsA(int Value) 
	{
		_X = Value;
	}


	void Print() 
	{
		cout << "Value Now:" << _X << endl;
	}



};
void Func1(clsA A1)
{
	A1._X = 100;
}
void Func2(clsA &A2)
{
	A2._X = 500;
}
int main()
{
	clsA A1(50);
	A1.Print();
	Func1(A1);
	A1.Print();
	Func2(A1);
	A1.Print();
	
	system("pause>0");
}
