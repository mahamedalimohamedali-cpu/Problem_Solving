short DayOfWeekOrder(short Year, short Month, short Day)
{
	int a = ((14 - Month) / 12);
	int Y = Year - a;
	int m = Month + (12 * a) - 2;

	return (Day + Y + (Y / 4) - (Y / 100) + (Y / 400) + ((31 * m) / 12)) % 7;

}
